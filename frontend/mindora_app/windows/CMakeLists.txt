cmake_minimum_required(VERSION 3.14)
project(mindora_app LANGUAGES CXX)

set(BINARY_NAME "mindora_app")
set(APPLICATION_ID "com.example.mindora_app")

cmake_policy(SET CMP0063 NEW)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)

get_filename_component(FLUTTER_MANAGED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/flutter" REALPATH)
set(EPHEMERAL_DIR "${FLUTTER_MANAGED_DIR}/ephemeral")

if(EXISTS "${EPHEMERAL_DIR}")
  add_subdirectory(${EPHEMERAL_DIR})
endif()

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE "Debug" CACHE STRING "" FORCE)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG "${PROJECT_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_PROFILE "${PROJECT_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${PROJECT_BINARY_DIR}")

add_subdirectory("runner")
